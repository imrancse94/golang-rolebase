package migrations

import "time"

type User struct {
	ID        uint      `gorm:"primaryKey"`
	Name      string    `gorm:"size:100;not null"`
	Email     string    `gorm:"size:255;index:idx_email,unique"`
	Password  string    `gorm:"size:255;not null"`
	CreatedAt time.Time `gorm:"type:timestamp;default:CURRENT_TIMESTAMP"`
	UpdatedAt time.Time `gorm:"type:timestamp;default:CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP"`
}

type Unit struct {
	ID        uint      `gorm:"primaryKey"`
	Label     string    `gorm:"size:50;not null"`
	Value     string    `gorm:"size:50;not null"`
	CreatedAt time.Time `gorm:"type:timestamp;default:CURRENT_TIMESTAMP"`
	UpdatedAt time.Time `gorm:"type:timestamp;default:CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP"`
}

type Product struct {
	ID             uint      `gorm:"primaryKey"`
	CategoryID     uint      `gorm:"not null"`
	Category       Category  `gorm:"foreignKey:CategoryID;references:ID"`
	CreatedBy      uint      `gorm:"not null"`
	Creator        User      `gorm:"foreignKey:CreatedBy;references:ID"`
	UpdatedBy      uint      `gorm:"not null"`
	Updater        User      `gorm:"foreignKey:UpdatedBy;references:ID"`
	Name           string    `gorm:"size:255;index:idx_email,unique"`
	UnitPrice      float64   `gorm:"type:decimal(10,2);not null;default:0.00"`
	Tax            float64   `gorm:"type:decimal(10,2);not null;default:0.00;comment:'Tax percentage'"`
	Discount       float64   `gorm:"type:decimal(10,2);not null;default:0.00;comment:'Discount percentage'"`
	AvailableStock int       `gorm:"not null;default:0"`
	Status         string    `gorm:"type:ENUM('active', 'inactive');default:'active'"`
	PromoCode      string    `gorm:"size:100;unique"`
	ImageURL       string    `gorm:"size:255;not null;comment:'Product into image URL'"`
	CreatedAt      time.Time `gorm:"type:timestamp;default:CURRENT_TIMESTAMP"`
	UpdatedAt      time.Time `gorm:"type:timestamp;default:CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP"`
}

type ProductDetails struct {
	ID          uint      `gorm:"primaryKey"`
	ProductID   uint      `gorm:"not null"`
	Product     Product   `gorm:"foreignKey:ProductID"`
	CreatedBy   uint      `gorm:"not null"`
	Creator     User      `gorm:"foreignKey:CreatedBy;references:ID"`
	UpdatedBy   uint      `gorm:"not null"`
	Updater     User      `gorm:"foreignKey:UpdatedBy;references:ID"`
	ImageURL    string    `gorm:"size:255;not null"`
	Description string    `gorm:"size:255;not null"`
	CreatedAt   time.Time `gorm:"type:timestamp;default:CURRENT_TIMESTAMP"`
	UpdatedAt   time.Time `gorm:"type:timestamp;default:CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP"`
}

type Category struct {
	ID        uint      `gorm:"primaryKey"`
	ParentID  uint      `gorm:"not null;default:0"`
	UnitID    uint      `gorm:"not null"`
	Unit      Unit      `gorm:"foreignKey:UnitID;references:ID"`
	Name      string    `gorm:"size:100;unique"`
	Status    string    `gorm:"type:ENUM('active', 'inactive');default:'active'"`
	CreatedAt time.Time `gorm:"type:timestamp;default:CURRENT_TIMESTAMP"`
	UpdatedAt time.Time `gorm:"type:timestamp;default:CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP"`
}

type Order struct {
	ID         uint      `gorm:"primaryKey"`
	UserID     uint      `gorm:"not null"`
	User       User      `gorm:"foreignKey:UserID;references:ID"`
	TotalPrice float64   `gorm:"type:decimal(10,2);not null;default:0.00"`
	Status     string    `gorm:"type:ENUM('pending', 'completed', 'canceled');default:'pending'"`
	CreatedAt  time.Time `gorm:"type:timestamp;default:CURRENT_TIMESTAMP"`
	UpdatedAt  time.Time `gorm:"type:timestamp;default:CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP"`
}

type OrderDetail struct {
	ID        uint      `gorm:"primaryKey"`
	OrderID   uint      `gorm:"not null"`

	UnitPrice float64   `gorm:"type:decimal(10,2);not null;default:0.00"`
	CreatedAt time.Time `gorm:"type:timestamp;default:CURRENT_TIMESTAMP"`
	UpdatedAt time.Time `gorm:"type:timestamp;default:CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP"`
}
